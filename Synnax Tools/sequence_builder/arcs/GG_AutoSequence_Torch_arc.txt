func check_redline() u8 {
	redline_count u8 := 0
	redline_count += PT_TI_01 < 402
	redline_count += PT_GG_01 < 497
	redline_count += PT_O2_04 < 500
	redline_count += PT_O2_05 < 610
	redline_count += PT_FU_01 < 550
	redline_count += PT_N2_08 < 1000
	redline_count += TC_GG_01 < 1868
	redline_count += TC_FU_02 < 300
	redline_count += TC_GO2_04 < 300
	redline_count += TC_GO2_05 < 300
	redline_count += TC_H2_03 < 300
	return redline_count
}
