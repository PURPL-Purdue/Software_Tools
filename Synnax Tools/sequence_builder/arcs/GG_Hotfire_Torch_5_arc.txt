func check_redline() u8 {
	redline_count u8 := 0
	redline_count += PT_TI_01 > 440
	redline_count += PT_GG_01 > 497
	redline_count += PT_H2_03 > 2100
	redline_count += PT_GO2_04_2 > 630
	redline_count += PT_GO2_05 > 610
	redline_count += PT_FU_02 > 550
	redline_count += PT_N2_08 > 1000
	redline_count += TC_GG_01 > 1868
	return redline_count
}
